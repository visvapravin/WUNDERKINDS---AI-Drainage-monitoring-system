# Example ESP32 IoT Program 
#include <WiFi.h>
#include <ESPmDNS.h>
#include <FirebaseESP32.h>
#include <TinyGPS++.h>
#include <HardwareSerial.h>

// WiFi Credentials (replace with your own in your private copy)
const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

// Firebase Configuration (replace with your own in your private copy)
#define FIREBASE_HOST "your-firebase-database-url"
#define FIREBASE_AUTH "your-firebase-auth-token"

// GPS Configuration
#define GPS_RX_PIN 16
#define GPS_TX_PIN 17
HardwareSerial SerialGPS(1);
TinyGPSPlus gps;

// Firebase Objects
FirebaseData fbdo;
FirebaseConfig config;
FirebaseAuth auth;

// Timing Variables
unsigned long lastSendTime = 0;
unsigned long lastWifiCheck = 0;
const unsigned long sendInterval = 5000;
const unsigned long wifiCheckInterval = 10000;

void debugWiFi() {
  Serial.println("\n=== WiFi Diagnostics ===");
  Serial.printf("MAC Address: %s\n", WiFi.macAddress().c_str());
  Serial.printf("Chip ID: %08X\n", (uint32_t)ESP.getEfuseMac());
  
  int8_t rssi = WiFi.RSSI();
  Serial.printf("Signal Strength: %d dBm\n", rssi);
  Serial.printf("Connection Status: %d\n", WiFi.status());
  
  Serial.println("Nearby Networks:");
  int n = WiFi.scanNetworks();
  for (int i = 0; i < n; i++) {
    Serial.printf("%d: %s (%d dBm) %s\n", 
      i+1, 
      WiFi.SSID(i).c_str(), 
      WiFi.RSSI(i),
      (WiFi.encryptionType(i) == WIFI_AUTH_OPEN) ? "open" : "secured");
  }
  Serial.println("=======================");
}

void setup() {
  Serial.begin(115200);
  SerialGPS.begin(9600, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
  delay(2000);

  // WiFi Configuration
  WiFi.mode(WIFI_STA);
  WiFi.disconnect(true);
  delay(100);
  WiFi.setAutoReconnect(true);
  WiFi.setSleep(false);
  WiFi.setTxPower(WIFI_POWER_19_5dBm);

  Serial.println("\nStarting Connection...");
  debugWiFi();

  WiFi.begin(ssid, password);
  
  unsigned long startTime = millis();
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
    
    if (millis() - startTime > 30000) {
      Serial.println("\nConnection Failed!");
      debugWiFi();
      
      switch(WiFi.status()) {
        case WL_NO_SSID_AVAIL:
          Serial.println("Error: SSID Not Found");
          break;
        case WL_CONNECT_FAILED:
          Serial.println("Error: Authentication Failed");
          break;
        case WL_CONNECTION_LOST:
          Serial.println("Error: Connection Lost");
          break;
        default:
          Serial.printf("Unknown Error Code: %d\n", WiFi.status());
      }
      
      Serial.println("Attempting recovery...");
      WiFi.disconnect(true);
      delay(1000);
      ESP.restart();
      return;
    }
  }
  
  Serial.println("\nConnected Successfully!");
  Serial.printf("IP Address: %s\n", WiFi.localIP().toString().c_str());

  // Firebase Initialization (Updated 2024 Method)
  config.database_url = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;
  
  // Dummy auth (required in newer versions)
  auth.user.email = "visvapravin265@gmail.com";
  auth.user.password = "Visva@321";

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);
  Firebase.setReadTimeout(fbdo, 1000 * 60);
  Firebase.setwriteSizeLimit(fbdo, "tiny");
  
  // Verify Firebase connection
  if (Firebase.ready()) {
    Serial.println("Firebase initialized successfully");
  } else {
    Serial.println("Firebase initialization failed");
  }
}

void loop() {
  // WiFi Maintenance
  if (millis() - lastWifiCheck > wifiCheckInterval) {
    if (WiFi.status() != WL_CONNECTED) {
      Serial.println("Connection Lost! Reconnecting...");
      WiFi.reconnect();
      delay(5000);
      
      if (WiFi.status() != WL_CONNECTED) {
        Serial.println("Reconnection Failed! Restarting...");
        ESP.restart();
      }
    }
    lastWifiCheck = millis();
  }

  // GPS Processing
  while (SerialGPS.available() > 0) {
    if (gps.encode(SerialGPS.read())) {
      if (millis() - lastSendTime > sendInterval && gps.location.isValid() && Firebase.ready()) {
        sendDataToFirebase();
        lastSendTime = millis();
      }
    }
  }

  // GPS Hardware Check
  if (millis() > 5000 && gps.charsProcessed() < 10) {
    Serial.println("No GPS detected - check wiring");
    delay(5000); // Non-blocking warning
  }

  delay(10);
}

void sendDataToFirebase() {
  String path = "/devices/esp32_" + String(ESP.getEfuseMac(), HEX) + "/" + String(millis());

  FirebaseJson json;
  json.add("lat", gps.location.lat());
  json.add("lng", gps.location.lng());
  json.add("speed", gps.speed.kmph());
  json.add("altitude", gps.altitude.meters());
  json.add("satellites", gps.satellites.value());
  json.add("timestamp", getFormattedDateTime());

  if (Firebase.setJSON(fbdo, path, json)) {
    Serial.println("Data sent to: " + fbdo.dataPath());
  } else {
    Serial.println("Failed: " + fbdo.errorReason());
    
    // Enhanced error handling
    if (fbdo.errorReason().indexOf("token") != -1) {
      Serial.println("Token error - verify FIREBASE_AUTH");
    }
  }
}

String getFormattedDateTime() {
  if (gps.date.isValid() && gps.time.isValid()) {
    char buffer[30];
    sprintf(buffer, "%04d-%02d-%02d %02d:%02d:%02d",
            gps.date.year(),
            gps.date.month(),
            gps.date.day(),
            gps.time.hour(),
            gps.time.minute(),
            gps.time.second());
    return String(buffer);
  }
  return "No GPS time";
}
